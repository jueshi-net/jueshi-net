import { useSearchParams } from 'next/navigation';
import { getTaskChain } from '@/lib/task-chain-api';

// 新增 effect
useEffect(() => {
  const taskId = searchParams.get('taskId');
  if (!taskId) return;
  
  const loadTaskChain = async () => {
    const taskChain = await getTaskChain(taskId);
    const context = taskChain.context;
    
    // 填充表单
    if (context.hsCode) {
      setLineItems([{
        description: `${context.hsCode} - ${context.productDescription}`,
        hsCode: context.hsCode,
      }]);
    }
    if (context.country || context.city) {
      setClientAddress([context.addressLine, context.city, context.province, context.postalCode, context.country].filter(Boolean).join(', '));
    }
  };
  
  loadTaskChain();
}, [searchParams]);

// src/app/api/task-chains/[id]/route.ts

export async function GET(_req: NextRequest, { params }: RouteParams) {
  const session = await auth();
  if (!session?.user?.id) {
    return NextResponse.json({ error: "未登录" }, { status: 401 });
  }
  
  const { id } = await params;
  const userId = session.user.id;
  
  const taskChain = await prisma.taskChainDraft.findFirst({
    where: { id, userId },  // ✅ 校验 userId
  });
  
  if (!taskChain) {
    return NextResponse.json({ error: "任务链不存在" }, { status: 404 });
  }
  
  return NextResponse.json({ taskChain });
}
