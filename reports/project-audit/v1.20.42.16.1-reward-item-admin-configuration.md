  name        String       // 显示名称
  description String?      // 说明
  costPoints  Int          // 兑换所需积分
  rewardType  String       // 奖励类型
  rewardValue Int          // 奖励数值
  enabled     Boolean      @default(true)
  sortOrder   Int          @default(0)
  createdAt   DateTime     @default(now())
  updatedAt   DateTime     @updatedAt
