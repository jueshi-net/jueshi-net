## 八、技术细节

### 脚本执行过程
```bash
# 1. 创建脚本
cat > scripts/create-beta-invites-v3.sh << 'EOF'
#!/bin/bash
# 使用 PM2 环境创建 Beta 邀请码
DB_URL=$(pm2 env 6 2>/dev/null | grep '^DATABASE_URL:' | sed 's/^DATABASE_URL: //')
psql "$DB_URL" <<'SQL'
INSERT INTO invite_codes (id, code, max_uses, used_count, is_active, expires_at, note, created_by, created_at)
VALUES 
  (gen_random_uuid()::text, 'BETA2026-001', 5, 0, true, '2026-07-18 23:59:59', 'Beta 测试批次 1', 'system', NOW()),
  (gen_random_uuid()::text, 'BETA2026-002', 5, 0, true, '2026-07-18 23:59:59', 'Beta 测试批次 2', 'system', NOW()),
  (gen_random_uuid()::text, 'BETA2026-003', 5, 0, true, '2026-07-18 23:59:59', 'Beta 测试批次 3', 'system', NOW())
ON CONFLICT (code) DO NOTHING;
SQL
EOF

# 2. 同步到 VPS
rsync -avz scripts/create-beta-invites-v3.sh deploy@192.129.155.149:/home/deploy/xixiong-saas/scripts/

# 3. 执行脚本
ssh deploy@192.129.155.149 "chmod +x /home/deploy/xixiong-saas/scripts/create-beta-invites-v3.sh && /home/deploy/xixiong-saas/scripts/create-beta-invites-v3.sh"

# 4. 验证结果
# 输出: 3 rows inserted, 验证通过

# 5. 清理临时脚本
rm -f scripts/create-beta-invites*.sh scripts/create-beta-invites*.js
```
