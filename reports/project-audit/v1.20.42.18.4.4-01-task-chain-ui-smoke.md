## 三、保存状态验证

### 3.1 自动保存机制
```typescript
const [saving, setSaving] = useState(false);
const [saveStatus, setSaveStatus] = useState<'idle' | 'saving' | 'saved' | 'error'>('idle');
const [lastSavedAt, setLastSavedAt] = useState<Date | null>(null);
const saveTimerRef = useRef<NodeJS.Timeout | null>(null);
const pendingSaveRef = useRef<Record<string, any> | null>(null);

// Debounced auto-save (1 second delay)
const autoSave = useCallback((patch: Record<string, any>) => {
  const newContext = { ...context, ...patch };
  setContext(newContext);
  
  // Store pending changes
  pendingSaveRef.current = newContext;
  setSaveStatus('saving');
  
  // Clear existing timer
  if (saveTimerRef.current) {
    clearTimeout(saveTimerRef.current);
  }
  
  // Set new timer (1 second debounce)
  saveTimerRef.current = setTimeout(() => {
    performSave(pendingSaveRef.current!);
  }, 1000);
}, [context]);
```

**验证结果：** ✅ 自动保存机制已实现（1 秒防抖）

### 3.4 保存失败处理
```typescript
const performSave = async (data: Record<string, any>) => {
  try {
    setSaving(true);
    const res = await fetch(`/api/task-chains/${taskId}`, {
      method: 'PATCH',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ context: data }),
    });
    
    if (!res.ok) throw new Error('Save failed');
    
    setSaveStatus('saved');
    setLastSavedAt(new Date());
    
    // Clear saved status after 2 seconds
    setTimeout(() => setSaveStatus('idle'), 2000);
  } catch (error) {
    setSaveStatus('error');
    console.error('Save error:', error);
  } finally {
    setSaving(false);
  }
};
```

**验证结果：** ✅ 保存失败处理已实现

### 7.1 API 权限检查
```typescript
export async function GET(_req: NextRequest, { params }: RouteParams) {
  const session = await auth();
  if (!session?.user?.id) {
    return NextResponse.json({ error: "未登录" }, { status: 401 });
  }

  const { id } = await params;
  const userId = session.user.id;

  const taskChain = await prisma.taskChainDraft.findFirst({
    where: { id, userId },  // 验证所有权
  });

  if (!taskChain) {
    return NextResponse.json({ error: "任务链不存在" }, { status: 404 });
  }

  return NextResponse.json({ taskChain });
}
```
