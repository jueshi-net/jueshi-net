## 一、设计目标

### 核心原则
1. **所有 active 奖励必须真实入账** - 不能出现虚假承诺
2. **事务一致性** - 积分扣减和权益发放必须原子性
3. **防重复兑换** - 库存、限兑、冷却期三重保护
4. **用户可见** - 兑换记录、权益状态必须对用户可见
5. **后台可管理** - 管理员可查看、审核、管理所有兑换

### 本轮范围
- ✅ 实现 growth_50 真实入账
- ✅ 实现 word_export_coupon 真实入账（或禁用）
- ✅ 实现 no_branding_coupon 真实入账（或禁用）
- ✅ ad_slot_7day 保持 inactive（不实现）
- ✅ 添加限制机制（库存、限兑、冷却期）
- ✅ 用户端兑换记录页面
- ✅ 后台兑换记录管理页面

## 四、限制机制设计

### 4.1 库存检查

```typescript
async function checkStock(rewardItem: RewardItem): Promise<{ ok: boolean; reason?: string }> {
  if (rewardItem.stockLimit === null) {
    return { ok: true }; // 无限制
  }
  
  if (rewardItem.stockUsed >= rewardItem.stockLimit) {
    return { ok: false, reason: '库存不足' };
  }
  
  return { ok: true };
}
```

### 4.2 限兑检查

```typescript
async function checkPerUserLimit(userId: string, rewardItem: RewardItem): Promise<{ ok: boolean; reason?: string }> {
  const userRewardCount = await prisma.userReward.count({
    where: {
      userId,
      rewardItemId: rewardItem.id,
    },
  });
  
  if (userRewardCount >= rewardItem.perUserLimit) {
    return { ok: false, reason: `每人限兑 ${rewardItem.perUserLimit} 次` };
  }
  
  return { ok: true };
}
```

### 4.3 冷却期检查

```typescript
async function checkCooldown(userId: string, rewardItem: RewardItem): Promise<{ ok: boolean; reason?: string }> {
  if (rewardItem.cooldownHours === 0) {
    return { ok: true }; // 无冷却期
  }
  
  const lastReward = await prisma.userReward.findFirst({
    where: {
      userId,
      rewardItemId: rewardItem.id,
    },
    orderBy: { createdAt: 'desc' },
  });
  
  if (lastReward) {
    const cooldownEnd = new Date(lastReward.createdAt.getTime() + rewardItem.cooldownHours * 60 * 60 * 1000);
    if (new Date() < cooldownEnd) {
      const remainingHours = Math.ceil((cooldownEnd.getTime() - Date.now()) / (60 * 60 * 1000));
      return { ok: false, reason: `冷却期未结束，还需 ${remainingHours} 小时` };
    }
  }
  
  return { ok: true };
}
```

### 4.4 时间窗口检查

```typescript
function checkTimeWindow(rewardItem: RewardItem): { ok: boolean; reason?: string } {
  const now = new Date();
  
  if (rewardItem.startAt && now < rewardItem.startAt) {
    return { ok: false, reason: '活动尚未开始' };
  }
  
  if (rewardItem.endAt && now > rewardItem.endAt) {
    return { ok: false, reason: '活动已结束' };
  }
  
  return { ok: true };
}
```

### 4.5 综合检查

```typescript
async function validateRedemption(userId: string, rewardItem: RewardItem): Promise<{ ok: boolean; reason?: string }> {
  // 1. 检查 enabled
  if (!rewardItem.enabled) {
    return { ok: false, reason: '奖励项已禁用' };
  }
  
  // 2. 检查时间窗口
  const timeCheck = checkTimeWindow(rewardItem);
  if (!timeCheck.ok) return timeCheck;
  
  // 3. 检查库存
  const stockCheck = await checkStock(rewardItem);
  if (!stockCheck.ok) return stockCheck;
  
  // 4. 检查限兑
  const perUserCheck = await checkPerUserLimit(userId, rewardItem);
  if (!perUserCheck.ok) return perUserCheck;
  
  // 5. 检查冷却期
  const cooldownCheck = await checkCooldown(userId, rewardItem);
  if (!cooldownCheck.ok) return cooldownCheck;
  
  // 6. 检查积分
  const user = await prisma.user.findUnique({ where: { id: userId } });
  if (!user || user.points < rewardItem.costPoints) {
    return { ok: false, reason: '积分不足' };
  }
  
  return { ok: true };
}
```

## 五、事务一致性设计

### 5.1 兑换事务

```typescript
async function redeemReward(userId: string, rewardItemCode: string) {
  return await prisma.$transaction(async (tx) => {
    // 1. 获取奖励项（带锁）
    const rewardItem = await tx.rewardItem.findUnique({
      where: { code: rewardItemCode },
    });
    
    if (!rewardItem) {
      throw new Error('奖励项不存在');
    }
    
    // 2. 验证兑换条件
    const validation = await validateRedemption(userId, rewardItem);
    if (!validation.ok) {
      throw new Error(validation.reason);
    }
    
    // 3. 扣减积分
    await tx.user.update({
      where: { id: userId },
      data: { points: { decrement: rewardItem.costPoints } },
    });
    
    // 4. 增加库存使用
    if (rewardItem.stockLimit !== null) {
      await tx.rewardItem.update({
        where: { id: rewardItem.id },
        data: { stockUsed: { increment: 1 } },
      });
    }
    
    // 5. 创建 UserReward
    const userReward = await tx.userReward.create({
      data: {
        userId,
        rewardItemId: rewardItem.id,
        rewardType: rewardItem.rewardType,
        rewardValue: rewardItem.rewardValue,
        status: rewardItem.requiresApproval ? 'pending' : 'active',
        pointsCost: rewardItem.costPoints,
        auditStatus: rewardItem.requiresApproval ? 'pending' : 'auto',
      },
    });
    
    // 6. 执行履约逻辑
    await fulfillReward(tx, userId, rewardItem, userReward);
    
    // 7. 记录积分变动
    await tx.pointLedger.create({
      data: {
        userId,
        type: 'reward_redeem',
        amount: -rewardItem.costPoints,
        reason: `兑换奖励: ${rewardItem.name}`,
        refType: 'user_reward',
        refId: userReward.id,
      },
    });
    
    return userReward;
  });
}
```

### 5.2 履约分发

```typescript
async function fulfillReward(
  tx: Prisma.TransactionClient,
  userId: string,
  rewardItem: RewardItem,
  userReward: UserReward
) {
  switch (rewardItem.rewardType) {
    case 'member_trial':
      await fulfillMemberTrial(tx, userId, rewardItem, userReward);
      break;
    case 'growth':
      await fulfillGrowth(tx, userId, rewardItem, userReward);
      break;
    case 'word_export_coupon':
      await fulfillWordExportCoupon(tx, userId, rewardItem, userReward);
      break;
    case 'no_branding_coupon':
      await fulfillNoBrandingCoupon(tx, userId, rewardItem, userReward);
      break;
    default:
      throw new Error(`未知的奖励类型: ${rewardItem.rewardType}`);
  }
}
```

### 9.2 数据回填

**回填脚本:**
```javascript
// 1. 为现有 reward items 设置默认值
await prisma.rewardItem.updateMany({
  data: {
    stockUsed: 0,
    perUserLimit: 1,
    cooldownHours: 0,
    requiresApproval: false,
  },
});

// 2. 为现有 user rewards 设置默认值
await prisma.userReward.updateMany({
  data: {
    auditStatus: 'auto',
    pointsCost: 0, // 需要根据 rewardItem.costPoints 回填
  },
});

// 3. 回填 pointsCost
const userRewards = await prisma.userReward.findMany();
for (const reward of userRewards) {
  const rewardItem = await prisma.rewardItem.findUnique({
    where: { id: reward.rewardItemId },
  });
  if (rewardItem) {
    await prisma.userReward.update({
      where: { id: reward.id },
      data: { pointsCost: rewardItem.costPoints },
    });
  }
}

// 4. 为 word_export_coupon 创建 CouponEntitlement
const wordExportRewards = await prisma.userReward.findMany({
  where: { rewardType: 'word_export_coupon' },
});

for (const reward of wordExportRewards) {
  await prisma.couponEntitlement.create({
    data: {
      userId: reward.userId,
      couponType: 'word_export',
      quantity: reward.rewardValue,
      status: 'active',
      expiresAt: new Date(Date.now() + 365 * 24 * 60 * 60 * 1000),
      sourceRewardId: reward.id,
    },
  });
}
```
