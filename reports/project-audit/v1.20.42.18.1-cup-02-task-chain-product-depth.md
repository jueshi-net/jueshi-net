const [searchQuery, setSearchQuery] = useState('');

const filteredChains = taskChains

### 2.3 重命名功能

**实现：**
- 点击"重命名"进入编辑模式
- 支持 Enter 保存、Escape 取消
- 调用 PATCH /api/task-chains/[id] 更新标题

**代码位置：**
```typescript
const handleRename = async (id: string) => {
  if (!renameValue.trim()) return;
  const res = await fetch(`/api/task-chains/${id}`, {
    method: 'PATCH',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ title: renameValue.trim() }),
  });
  if (res.ok) {
    setTaskChains(prev => prev.map(tc => tc.id === id ? { ...tc, title: renameValue.trim() } : tc));
  }
};
```

### 2.4 复制任务功能

**实现：**
- 点击"复制任务"创建副本
- 副本标题添加 "(副本)" 后缀
- 重置步骤进度（currentStep=0, completedSteps=[]）
- 保留其他上下文数据

**代码位置：**
```typescript
const handleDuplicate = async (task: TaskChainDraft) => {
  const res = await fetch('/api/task-chains', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      title: `${task.title} (副本)`,
      sourceTool: task.sourceTool,
      context: {
        ...task.context,
        currentStep: 0,
        completedSteps: [],
      },
    }),
  });
};
```
