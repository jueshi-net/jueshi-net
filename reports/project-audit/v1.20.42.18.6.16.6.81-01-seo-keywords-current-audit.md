### 4. 页面 metadata 输出

#### Checklist 页面 (`src/app/(public)/checklists/[slug]/page.tsx`)

```typescript
// v1.20.42.18.6.16.6.80: Extract keywords from metadataJson
let keywords = "";
if (c.metadataJson) {
  const meta = JSON.parse(c.metadataJson);
  const contentOps = meta.contentOps || {};
  const keywordSet = new Set<string>();
  
  // Primary keyword
  if (contentOps.primaryKeyword) {
    keywordSet.add(contentOps.primaryKeyword);
  }
  
  // Secondary keywords
  if (Array.isArray(contentOps.secondaryKeywords)) {
    contentOps.secondaryKeywords.forEach((kw) => keywordSet.add(kw));
  }
  
  // Page type keyword
  keywordSet.add("清单");
  
  // Limit to 15 keywords
  const keywordArray = Array.from(keywordSet).slice(0, 15);
  keywords = keywordArray.join(",");
}

// Fallback
if (!keywords) {
  keywords = `${c.title},清单,出国清单,留学清单,checklist`;
}

return {
  title: c.seoTitle || c.title,
  description: c.seoDescription || c.summary || "",
  keywords,
  alternates: { canonical: c.canonicalUrl || `https://jueshi.net/checklists/${c.slug}` },
  robots: (previewMode || isDraft) ? { index: false, follow: false } : undefined,
};
```
